Code blocks are powered by **Shiki** for syntax highlighting. They support
titles, line numbers, line highlighting, diffs, and more.

## Basic code block

```ts
function greet(name: string): string {
  return `Hello, ${name}!`;
}
```

## With title

```ts title="src/lib/greet.ts"
function greet(name: string): string {
  return `Hello, ${name}!`;
}
```

## Line highlighting

```ts {2-3}
function greet(name: string): string {
  // highlighted lines
  const message = `Hello, ${name}!`;
  return message;
}
```

## Word highlighting

```ts /greet/
function greet(name: string): string {
  return `Hello, ${name}!`;
}

const result = greet("World");
```

## Diff

```ts
function greet(name: string): string {
- return `Hello!`;
+ return `Hello, ${name}!`;
}
```

## Multiple features combined

```ts title="src/lib/utils.ts" {3} /transform/
function transform(input: string): string {
  const trimmed = input.trim();
  const result = trimmed.toUpperCase(); // highlighted
  return result;
}
```
