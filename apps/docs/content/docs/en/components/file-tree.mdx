The `Files` component renders interactive file trees. Folders are collapsible.

## Basic file tree

<Files>
	<Folder name="src" defaultOpen>
		<File name="index.ts" />
		<File name="utils.ts" />
		<Folder name="components">
			<File name="Button.tsx" />
			<File name="Card.tsx" />
		</Folder>
	</Folder>
	<File name="package.json" />
	<File name="tsconfig.json" />
</Files>

## Fuutu Stack structure

<Files>
	<Folder name="apps/docs" defaultOpen>
		<Folder name="content/docs" defaultOpen>
			<Folder name="en" defaultOpen>
				<File name="meta.json" />
				<File name="index.mdx" />
				<Folder name="getting-started">
					<File name="meta.json" />
					<File name="introduction.mdx" />
				</Folder>
			</Folder>
		</Folder>
		<Folder name="src">
			<Folder name="app">
				<File name="layout.tsx" />
				<Folder name="[locale]">
					<File name="layout.tsx" />
					<File name="page.tsx" />
				</Folder>
			</Folder>
			<Folder name="lib">
				<File name="source.ts" />
				<File name="i18n.ts" />
			</Folder>
		</Folder>
		<File name="package.json" />
		<File name="source.config.ts" />
	</Folder>
</Files>

## Usage

```mdx
<Files>
  <Folder name="src" defaultOpen>
    <File name="index.ts" />
    <Folder name="components">
      <File name="Button.tsx" />
    </Folder>
  </Folder>
  <File name="package.json" />
</Files>
```
